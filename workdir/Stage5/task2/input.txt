decl
    int factorial(int n);
enddecl

int factorial(int n) {
    decl
        int f;
    enddecl
    begin
        if (n == 0 || n == 1) then
            f = 1;
        else
            f = n * factorial(n - 1);
        endif;
        return f;
    end
}

int main() {
    decl
        int res;
    enddecl
    begin
        res = factorial(5);
        write(res);
        return 0;
    end
}
